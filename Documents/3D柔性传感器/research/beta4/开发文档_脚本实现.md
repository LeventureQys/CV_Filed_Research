### 3.2 filters.py（beta4 版本）

```python
from beta3.scripts.filters import *  # 引用所有 beta3 算法

def adaptive_baseline(cells: np.ndarray, sample_rate: float, window_s: float = 10.0) -> np.ndarray:
    """自适应 Cell 级基线估计与扣除。"""
    window = max(1, round(sample_rate * window_s))
    out = np.empty_like(cells)
    for i in range(len(cells)):
        left = max(0, i - window)
        # 对每个 Cell 取历史窗口内的中位数
        out[i] = cells[i] - np.median(cells[left:i+1], axis=0) if i >= window else cells[i]
    return np.maximum(out, 0.0)

def dual_state_kalman(values: np.ndarray, sample_rate: float,
                      q_force: float = 0.01, q_drift: float = 1e-4,
                      r_measure: float = 1.0) -> np.ndarray:
    """双状态卡尔曼：总力 + 慢漂移。"""
    dt = 1.0 / max(sample_rate, 1e-6)
    F = np.array([[1.0, dt], [0.0, 1.0]])
    H = np.array([[1.0, 0.0]])
    Q = np.diag([q_force, q_drift])
    R = np.array([[r_measure]])
    
    x = np.array([values[0], 0.0])  # 初始力值，漂移为 0
    P = np.eye(2) * r_measure
    out = np.empty_like(values)
    out[0] = x[0]
    
    for k in range(1, len(values)):
        x = F @ x
        P = F @ P @ F.T + Q
        S = H @ P @ H.T + R
        K = P @ H.T / S[0, 0]
        x = x + K.flatten() * (values[k] - (H @ x)[0])
        P = (np.eye(2) - np.outer(K, H[0])) @ P
        out[k] = x[0]
    return out

def huber_ema(values: np.ndarray, sample_rate: float,
              time_constant: float = 0.8,
              threshold_sigma: float = 3.0) -> np.ndarray:
    """Huber 加权鲁棒 EMA。"""
    alpha = 1.0 - np.exp(-1.0 / (sample_rate * time_constant))
    out = np.empty_like(values)
    out[0] = values[0]
    residual_history = []  # 滚动 MAD 估计
    
    for k in range(1, len(values)):
        residual = values[k] - out[k-1]
        residual_history.append(abs(residual))
        if len(residual_history) > max(50, round(sample_rate * 5)):
            residual_history.pop(0)
        
        # 在线 Sigma 估计：MAD * 1.4826
        sigma = np.median(residual_history) * 1.4826 if len(residual_history) > 5 else abs(residual)
        sigma = max(sigma, 1e-6)
        
        # Huber 截断
        if abs(residual) > threshold_sigma * sigma:
            residual = np.sign(residual) * threshold_sigma * sigma
        
        out[k] = out[k-1] + alpha * residual
    return out
```
